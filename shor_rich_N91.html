Код алгоритма
def shor_factorize(N, max_attempts=10):
    for attempt in range(max_attempts):
        a = random.randint(2, N-1)
        g = gcd(a, N)
        if g != 1:
            return g, N//g  # trivial factor
        
        r = find_period(a, N)  # quantum period finding
        if r is None or r % 2 != 0:
            continue
        
        x = pow(a, r//2, N)
        if x == N - 1:
            continue
        
        f1 = gcd(x-1, N)
        f2 = gcd(x+1, N)
        if f1 not in (1, N):
            return f1, f2
    return None

def find_period(a, N):
    for r in range(1, N):
        if pow(a, r, N) == 1:
            return r
    return None
